// Import the `useState` hook from React to manage state in this functional component.
import { useState } from "react";

// Define the main component for the registration form.
export default function RegistrationForm() {
  // Initialize state using `useState`. `formData` holds the form values, and `setFormData` updates them.
  const [formData, setFormData] = useState({
    association: "", // Stores the selected association from the dropdown.
    secondDropdown: "", // Stores the selected option from the second dropdown.
    name: "", // Stores the user's full name.
    dob: "", // Stores the user's date of birth.
    role: "", // Stores the user's position or role.
  });

  // Function to handle changes in the form inputs.
  const handleChange = (e) => {
    // Update the state by spreading the existing `formData` and overwriting the changed field.
    setFormData({ ...formData, [e.target.name]: e.target.value });
  };

  // Function to handle form submission.
  const handleSubmit = (e) => {
    e.preventDefault(); // Prevent the default form submission behavior (page reload).
    console.log(formData); // Log the form data to the console (for now).
  };

  // Render the form UI using JSX.
  return (
    // Container for the form with Tailwind CSS classes for styling.
    <div className="max-w-lg mx-auto mt-10 p-8 bg-white shadow-lg rounded-xl border border-gray-200 md:max-w-2xl lg:max-w-3xl xl:max-w-4xl">
      {/* Form title */}
      <h2 className="text-3xl font-bold text-gray-800 mb-6 text-center">Register</h2>

      {/* Form element with `onSubmit` handler */}
      <form onSubmit={handleSubmit} className="space-y-6">
        {/* Association Dropdown */}
        <div>
          <label className="block text-sm font-semibold text-gray-700">Select Association</label>
          <select
            name="association" // Matches the key in `formData`.
            value={formData.association} // Tied to the state.
            onChange={handleChange} // Calls `handleChange` when the user selects an option.
            className="mt-2 block w-full p-3 border border-gray-300 rounded-lg focus:ring-2 focus:ring-blue-500 focus:outline-none"

            <option value="limbe-north">Limbe North Association</option>

          >

            <option value="option2">Option 2</option>
          </select>
        </div>

        {/* Name Input */}
        <div>
          <label className="block text-sm font-semibold text-gray-700">Full Name</label>
          <input
            type="text"
            name="name" // Matches the key in `formData`.
            value={formData.name} // Tied to the state.
            onChange={handleChange} // Calls `handleChange` when the user types.
            className="mt-2 block w-full p-3 border border-gray-300 rounded-lg focus:ring-2 focus:ring-blue-500 focus:outline-none"
            placeholder="Enter your name"
          />
        </div>

        {/* Date of Birth Input */}
        <div>
          <label className="block text-sm font-semibold text-gray-700">Date of Birth</label>
          <input
            type="date"
            name="dob" // Matches the key in `formData`.
            value={formData.dob} // Tied to the state.
            onChange={handleChange} // Calls `handleChange` when the user selects a date.
            className="mt-2 block w-full p-3 border border-gray-300 rounded-lg focus:ring-2 focus:ring-blue-500 focus:outline-none"
